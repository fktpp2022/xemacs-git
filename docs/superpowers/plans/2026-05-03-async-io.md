# XEmacs Async IO Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add cooperative coroutine scheduling, libcurl-based async HTTP with SSE streaming, and an actor model to XEmacs core — enabling a mini-SWE-agent that talks to LLMs and executes tool calls in parallel.

**Architecture:** Two new C files (`async-scheduler.c`, `async-http.c`) plus a portable assembly stack-swap (`coro-asm.S`) are added to `src/`. Two targeted patches to `event-unixoid.c` expose fd registration and wakeup APIs. Four Elisp files (`async-core.el`, `actor.el`, `async-http.el`, `async-file.el`) build the user-facing API. Coroutines are cooperative: only one runs at a time, yielding at `await` points, with GC roots maintained manually through the scheduler.

**Tech Stack:** C (C99), POSIX `ucontext_t` / x86-64+ARM64 assembly fallback, libcurl multi interface, XEmacs Lisp test-harness (`Assert`, `Check-Error`), CMake, Python 3 (test HTTP server).

## Task 3: `coro-asm.S` — portable stack-swap primitives

This replaces `ucontext_t`'s `makecontext`/`swapcontext` with raw assembly that works identically on Linux and macOS (where `ucontext_t` is deprecated).

**Files:**
- Modify: `src/coro-asm.S`

- [ ] **Step 1: Write the assembly**

Replace the stub `src/coro-asm.S` with:

```asm
/* Portable coroutine stack-swap for XEmacs async scheduler.
   Implements coro_init_stack and coro_swap.
   Based on the approach used by LuaJIT/libco. */

#if defined(__x86_64__)

/* struct coro_context { void *rsp; }; — first field is saved stack pointer */

/* void coro_swap(coro_context *from, coro_context *to)
   rdi = from, rsi = to */
    .text
    .globl coro_swap
    .type  coro_swap, @function
coro_swap:
    /* Save callee-saved registers + return address onto current stack */
    pushq %rbp
    pushq %rbx
    pushq %r12
    pushq %r13
    pushq %r14
    pushq %r15
    /* Save current stack pointer into from->rsp */
    movq  %rsp, (%rdi)
    /* Restore stack pointer from to->rsp */
    movq  (%rsi), %rsp
    /* Restore callee-saved registers */
    popq  %r15
    popq  %r14
    popq  %r13
    popq  %r12
    popq  %rbx
    popq  %rbp
    /* Return into the resumed coroutine */
    ret
    .size coro_swap, .-coro_swap

#elif defined(__aarch64__)

    .text
    .globl coro_swap
    .type  coro_swap, %function
coro_swap:
    /* Save callee-saved registers to current stack */
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!
    stp x29, x30, [sp, #-16]!
    /* Save sp into from->sp */
    mov x2, sp
    str x2, [x0]
    /* Load sp from to->sp */
    ldr x2, [x1]
    mov sp, x2
    /* Restore callee-saved registers */
    ldp x29, x30, [sp], #16
    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
    .size coro_swap, .-coro_swap

#else
/* Fallback: use ucontext_t on other architectures */
#endif
```

/* async-scheduler.h — XEmacs cooperative coroutine scheduler */
#ifndef XEMACS_ASYNC_SCHEDULER_H
#define XEMACS_ASYNC_SCHEDULER_H

#include "lisp.h"

/* Coroutine state machine */
typedef enum {
  CORO_RUNNABLE  = 0,
  CORO_RUNNING   = 1,
  CORO_SUSPENDED = 2,
  CORO_DEAD      = 3
} coro_state_t;

/* Suspension reason — determines what will wake this coroutine */
typedef enum {
  WAIT_NONE     = 0,  /* runnable */
  WAIT_FD       = 1,  /* waiting for an fd to become ready */
  WAIT_TIMER    = 2,  /* waiting for a timeout */
  WAIT_MAILBOX  = 3,  /* waiting for an actor message */
  WAIT_JOIN     = 4,  /* waiting for another coroutine to finish */
} wait_reason_t;

struct xemacs_coro;

/* Context: just the saved stack pointer (coro_swap uses this) */
typedef struct {
  void *sp;
} coro_context_t;

typedef struct xemacs_coro {
  coro_context_t  ctx;           /* saved CPU state (stack pointer) */
  coro_context_t  caller_ctx;    /* scheduler's context to return to */
  unsigned char  *stack;         /* malloc'd coroutine stack */
  Bytecount       stack_size;    /* stack size in bytes */
  coro_state_t    state;
  wait_reason_t   wait_reason;

  /* Result/error delivered on resume */
  Lisp_Object     result;
  Lisp_Object     error;         /* non-nil → re-signal on resume */

  /* GC: Lisp objects live on the C stack — must be rooted manually */
  struct gcpro   *gcpro_chain;   /* chain head for this coro's live objects */

  /* Suspension details */
  int             wait_fd;       /* fd we're blocked on (WAIT_FD) */
  EMACS_TIME      wait_deadline; /* absolute timeout (WAIT_TIMER) */

  /* Actor mailbox */
  Lisp_Object     mailbox;       /* list of pending messages */
  Lisp_Object     actor_name;    /* symbol or nil */
  int             is_actor;

  /* Join: other coroutines waiting on this one */
  struct xemacs_coro *join_waiter;

  /* Linked list of all coroutines */
  struct xemacs_coro *next;
  struct xemacs_coro *prev;
} xemacs_coro;

/* Assembly primitive — swaps stack pointer between two coro contexts */
extern void coro_swap (coro_context_t *from, coro_context_t *to);

/* Lifecycle */
extern xemacs_coro *coro_spawn (Lisp_Object (*fn)(Lisp_Object), Lisp_Object arg);
extern void         coro_yield (wait_reason_t reason);
extern void         coro_resume (xemacs_coro *c, Lisp_Object result);
extern void         coro_resume_with_error (xemacs_coro *c, Lisp_Object error);

/* Scheduler tick — called by event-unixoid.c after select() */

                                          Lisp_Object args);
extern void         actor_send_internal (xemacs_coro *target, Lisp_Object msg);

extern void vars_of_async_scheduler (void);

- [ ] **Step 2: Write `async-scheduler.c` — data and lifecycle**

Replace the stub with the full implementation. Write in sections:

```c
/* async-scheduler.c — XEmacs cooperative coroutine scheduler */
#include <config.h>
#include "lisp.h"
#include "async-scheduler.h"
#include "events.h"
#include "systime.h"
#include "syssignal.h"
#include <assert.h>
#include <stdlib.h>
#include <string.h>

/* Default coroutine stack size: 256KB */
#define CORO_DEFAULT_STACK_SIZE (256 * 1024)

/* The currently running coroutine (NULL = main Lisp thread) */
static xemacs_coro *current_coro = NULL;

/* Doubly-linked list of all live coroutines */
static xemacs_coro *all_coros = NULL;

/* Scheduler's own context — the "main" context we return to after each tick */
static coro_context_t scheduler_ctx;

/* Number of live (non-DEAD) coroutines */
static int coro_count = 0;

Lisp_Object Qasync_timeout;
Lisp_Object Qasync_error;

/* Lisp variable: default stack size */
static Fixnum Vasync_coroutine_stack_size;

/* ---- Intrusive list helpers ---- */

static void
coro_list_add (xemacs_coro *c)
{
  c->next = all_coros;
  c->prev = NULL;
  if (all_coros) all_coros->prev = c;
  all_coros = c;
  coro_count++;
}

static void
coro_list_remove (xemacs_coro *c)
{
  if (c->prev) c->prev->next = c->next;
  else all_coros = c->next;
  if (c->next) c->next->prev = c->prev;
  coro_count--;
}

/* ---- Trampoline: entry point for a new coroutine ---- */
/*
 * When coro_swap transfers control to a new coroutine for the first time,
 * it lands here.  'current_coro->result' holds the argument passed to spawn.
 * When the function returns, we mark the coro DEAD and swap back to scheduler.
 */
static void
coro_trampoline (void)
{
  xemacs_coro *c = current_coro;
  Lisp_Object arg = c->result;
  Lisp_Object ret = Qnil;

  /* The coroutine's function is stored in c->error field temporarily
     during spawn (before first resume). Retrieve and clear it. */
  Lisp_Object fn = c->error;
  c->error = Qnil;

  /* Run the function, catch any unhandled signals */
  {
    int speccount = specpdl_depth ();
    condition_case_1 (Qt,
                      ret = call1 (fn, arg),
                      Qerror,
                      ret = Qnil;
                      if (c->join_waiter)
                        coro_resume_with_error (c->join_waiter, data);
                      c->join_waiter = NULL);
    unbind_to (speccount, Qnil);
  }

  /* Normal completion: wake any join_waiter */
  c->result = ret;
  if (c->join_waiter)
    {
      coro_resume (c->join_waiter, ret);
      c->join_waiter = NULL;
    }

  c->state = CORO_DEAD;
  coro_list_remove (c);

  /* Return to scheduler */
  coro_swap (&c->ctx, &scheduler_ctx);
  /* Never reached */
}

/* ---- spawn ---- */

xemacs_coro *
coro_spawn (Lisp_Object (*fn)(Lisp_Object), Lisp_Object arg)
{
  xemacs_coro *c = xnew_and_zero (xemacs_coro);
  Bytecount stack_size = Vasync_coroutine_stack_size > 0
    ? Vasync_coroutine_stack_size
    : CORO_DEFAULT_STACK_SIZE;

  c->stack = (unsigned char *) xmalloc (stack_size);
  c->stack_size = stack_size;
  c->state = CORO_RUNNABLE;
  c->wait_reason = WAIT_NONE;
  c->wait_fd = -1;
  c->mailbox = Qnil;
  c->result = arg;
  c->error = Qnil;   /* will be set below */
  c->gcpro_chain = NULL;
  c->join_waiter = NULL;
  c->is_actor = 0;
  c->actor_name = Qnil;

  /* Set up stack: stack grows down, sp starts near top of allocated block.
     Leave 128 bytes of red-zone on x86-64. */
  {
    unsigned char *stack_top = c->stack + stack_size - 128;
    /* Align to 16 bytes */
    stack_top = (unsigned char *)((uintptr_t)stack_top & ~(uintptr_t)15);
    /* Push the trampoline address as the return address */
    stack_top -= sizeof (void *);
    *(void **)stack_top = (void *)coro_trampoline;
    /* coro_swap will pop callee-saved regs; pre-allocate their space */
    stack_top -= 6 * sizeof (void *); /* rbp, rbx, r12-r15 on x86-64 */
    memset (stack_top, 0, 6 * sizeof (void *));
    c->ctx.sp = stack_top;
  }

  /* Store fn in c->error temporarily — trampoline picks it up */
  c->error = make_opaque_ptr ((void *)fn);

  coro_list_add (c);
  return c;
}

/* ---- yield ---- */

void
coro_yield (wait_reason_t reason)
{
  xemacs_coro *c = current_coro;
  assert (c != NULL);
  c->state = CORO_SUSPENDED;
  c->wait_reason = reason;
  current_coro = NULL;
  coro_swap (&c->ctx, &scheduler_ctx);
  /* Resumes here when woken. current_coro has been set back to c by
     async_scheduler_tick before the swap. */
}

/* ---- resume ---- */

void
coro_resume (xemacs_coro *c, Lisp_Object result)
{
  c->result = result;
  c->error = Qnil;
  c->state = CORO_RUNNABLE;
  c->wait_reason = WAIT_NONE;
}

void
coro_resume_with_error (xemacs_coro *c, Lisp_Object error)
{
  c->result = Qnil;
  c->error = error;
  c->state = CORO_RUNNABLE;
  c->wait_reason = WAIT_NONE;
}

/* ---- Scheduler tick ---- */

void
async_scheduler_tick (void)
{
  xemacs_coro *c, *next;
  EMACS_TIME now;

  if (!all_coros) return;

  EMACS_GET_TIME (now);

  /* Phase 1: advance timer-waiting coroutines */
  for (c = all_coros; c; c = c->next)
    {
      if (c->state == CORO_SUSPENDED && c->wait_reason == WAIT_TIMER)
        {
          if (EMACS_TIME_LE (c->wait_deadline, now))
            coro_resume_with_error (c, list2 (Qasync_timeout,
                                              build_string ("timeout")));
        }
    }

  /* Phase 2: run all RUNNABLE coroutines */
  for (c = all_coros; c; c = next)
    {
      next = c->next;   /* cache: c may be removed during run */
      if (c->state != CORO_RUNNABLE) continue;

      c->state = CORO_RUNNING;
      current_coro = c;

      /* If the coro has a pending error, signal it on resume */
      /* (handled inside coro_trampoline via the condition_case) */

      coro_swap (&scheduler_ctx, &c->ctx);
      /* We're back in the scheduler */
      current_coro = NULL;

      if (c->state == CORO_DEAD)
        {
          /* Free stack */
          xfree (c->stack);
          xfree (c);
        }
    }
}

/* ---- GC integration ---- */

void
async_scheduler_mark_gcpros (void)
{
  xemacs_coro *c;
  for (c = all_coros; c; c = c->next)
    {
      /* Mark mailbox */
      mark_object (c->mailbox);
      mark_object (c->result);
      mark_object (c->error);
      mark_object (c->actor_name);
      /* Walk the gcpro chain the coroutine left on its stack */
      {
        struct gcpro *p = c->gcpro_chain;
        while (p)
          {
            int i;
            for (i = 0; i < p->nvars; i++)
              mark_object (p->var[i]);
            p = p->next;
          }
      }
    }
}

/* ---- Actor primitives ---- */

xemacs_coro *
actor_spawn_internal (Lisp_Object name, Lisp_Object fn, Lisp_Object args)
{
  xemacs_coro *c = coro_spawn (
    (Lisp_Object (*)(Lisp_Object)) apply1,   /* fn(args) via apply1 */
    list2 (fn, args));
  c->is_actor = 1;
  c->actor_name = name;
  return c;
}

void
actor_send_internal (xemacs_coro *target, Lisp_Object msg)
{
  /* Append to mailbox */
  target->mailbox = nconc2 (target->mailbox, list1 (msg));
  /* If the actor is waiting for a message, wake it */
  if (target->state == CORO_SUSPENDED &&
      target->wait_reason == WAIT_MAILBOX)
    coro_resume (target, msg);
}

Lisp_Object
actor_receive_internal (int timeout_ms)
{
  xemacs_coro *c = current_coro;
  assert (c != NULL && c->is_actor);

  if (!NILP (c->mailbox))
    {
      Lisp_Object msg = XCAR (c->mailbox);
      c->mailbox = XCDR (c->mailbox);
      return msg;
    }

  /* Set timeout if requested */
  if (timeout_ms > 0)
    {
      EMACS_GET_TIME (c->wait_deadline);
      EMACS_SET_SECS_USECS (c->wait_deadline,
                             EMACS_SECS (c->wait_deadline) + timeout_ms / 1000,
                             EMACS_USECS (c->wait_deadline) + (timeout_ms % 1000) * 1000);
    }

  coro_yield (timeout_ms > 0 ? WAIT_TIMER : WAIT_MAILBOX);

  /* After resume: result holds the message */
  return c->result;
}

/* async-http.c — libcurl multi-handle async HTTP for XEmacs */
#include <config.h>
#include "lisp.h"
#include "async-http.h"
#include "events.h"
#include <stdlib.h>
#include <string.h>

#ifdef HAVE_LIBCURL

static CURLM *curl_multi = NULL;

Lisp_Object Qasync_http_error;

/* ---- Write callback: called by libcurl for each received chunk ---- */

static size_t
http_write_callback (char *ptr, size_t size, size_t nmemb, void *userdata)
{
  async_http_req *req = (async_http_req *) userdata;
  size_t nbytes = size * nmemb;

  if (!NILP (req->on_chunk))
    {
      /* Streaming mode: deliver chunk to Elisp callback immediately */
      Lisp_Object chunk = make_extstring ((Extbyte *) ptr, nbytes,
                                          Qutf_8);
      enqueue_misc_user_event (Qnil, req->on_chunk, chunk);
      signal_async_wakeup ();
    }
  else
    {
      /* Buffered mode: append to response_buf */
      Lisp_Object chunk = make_extstring ((Extbyte *) ptr, nbytes,
                                          Qutf_8);
      /* Concatenate into response_buf */
      req->response_buf = concat2 (req->response_buf, chunk);
    }

  return nbytes;
}

/* ---- Header callback ---- */

static size_t
http_header_callback (char *ptr, size_t size, size_t nmemb, void *userdata)
{
  async_http_req *req = (async_http_req *) userdata;
  size_t nbytes = size * nmemb;
  /* Parse "Name: Value\r\n" lines into headers_alist */
  if (nbytes > 2 && ptr[nbytes-2] == '\r')
    {
      char *colon = memchr (ptr, ':', nbytes);
      if (colon)
        {
          Lisp_Object name = make_extstring ((Extbyte *) ptr,
                                             colon - ptr, Qutf_8);
          char *val = colon + 1;
          while (*val == ' ') val++;
          size_t vlen = nbytes - (val - ptr) - 2; /* strip \r\n */
          Lisp_Object value = make_extstring ((Extbyte *) val, vlen,
                                              Qutf_8);
          req->headers_alist = Fcons (Fcons (name, value),
                                      req->headers_alist);
        }
    }
  return nbytes;
}

/* ---- Socket callback: called by libcurl to register/unregister fds ---- */

static int
curl_socket_callback (CURL *easy, curl_socket_t s, int action,
                      void *userp, void *socketp)
{
  (void) easy; (void) userp; (void) socketp;

  if (action == CURL_POLL_REMOVE)
    remove_extra_fd ((int) s);
  else
    add_extra_fd ((int) s, async_http_socket_ready, NULL);

  return 0;
}

/* ---- Timer callback: libcurl wants us to call curl_multi_socket_action ---- */

static int
curl_timer_callback (CURLM *multi, long timeout_ms, void *userp)
{
  (void) multi; (void) userp;
  /* If timeout_ms == 0, fire immediately; if -1, cancel timer.
     We handle this in async_http_tick via curl_multi_timeout. */
  (void) timeout_ms;
  return 0;
}

/* ---- Socket ready callback (called from extra-fd dispatch) ---- */

void
async_http_socket_ready (int fd, void *data)
{
  (void) data;
  int running;
  curl_multi_socket_action (curl_multi, (curl_socket_t) fd,
                             CURL_CSELECT_IN | CURL_CSELECT_OUT, &running);
  async_http_check_completed ();
}

/* Check for completed transfers and wake waiting coroutines */
void
async_http_check_completed (void)
{
  CURLMsg *msg;
  int msgs_left;

  while ((msg = curl_multi_info_read (curl_multi, &msgs_left)) != NULL)
    {
      if (msg->msg == CURLMSG_DONE)
        {
          CURL *easy = msg->easy_handle;
          async_http_req *req = NULL;
          curl_easy_getinfo (easy, CURLINFO_PRIVATE, &req);

          if (!req) continue;

          curl_easy_getinfo (easy, CURLINFO_RESPONSE_CODE, &req->status_code);
          req->curl_result = msg->data.result;

          curl_multi_remove_handle (curl_multi, easy);
          curl_easy_cleanup (easy);
          if (req->req_headers)
            curl_slist_free_all (req->req_headers);

          if (req->waiting_coro)
            {
              if (req->curl_result == CURLE_OK)
                coro_resume (req->waiting_coro,
                             list3 (make_fixnum (req->status_code),
                                    req->headers_alist,
                                    req->response_buf));
              else
                coro_resume_with_error (
                  req->waiting_coro,
                  list2 (Qasync_http_error,
                         build_extstring (curl_easy_strerror (req->curl_result),
                                          Qutf_8)));
              req->waiting_coro = NULL;
            }

          xfree (req);
        }
    }
}

/* ---- Tick: called each scheduler tick ---- */

void
async_http_tick (void)
{
  int running;
  long timeout_ms;

  if (!curl_multi) return;

  curl_multi_timeout (curl_multi, &timeout_ms);
  if (timeout_ms == 0)
    curl_multi_socket_action (curl_multi, CURL_SOCKET_TIMEOUT, 0, &running);

  async_http_check_completed ();
}

/* ---- Start a request ---- */

Lisp_Object
async_http_start (const char *url, const char *method,
                  Lisp_Object req_headers_alist,
                  Lisp_Object body_string,
                  Lisp_Object on_chunk,
                  int timeout_ms)
{
  async_http_req *req = xnew_and_zero (async_http_req);
  CURL *easy = curl_easy_init ();

  req->easy = easy;
  req->response_buf = build_string ("");
  req->on_chunk = on_chunk;
  req->headers_alist = Qnil;
  req->waiting_coro = current_coro;

  curl_easy_setopt (easy, CURLOPT_URL, url);
  curl_easy_setopt (easy, CURLOPT_WRITEFUNCTION, http_write_callback);
  curl_easy_setopt (easy, CURLOPT_WRITEDATA, req);
  curl_easy_setopt (easy, CURLOPT_HEADERFUNCTION, http_header_callback);
  curl_easy_setopt (easy, CURLOPT_HEADERDATA, req);
  curl_easy_setopt (easy, CURLOPT_PRIVATE, req);
  curl_easy_setopt (easy, CURLOPT_FOLLOWLOCATION, 1L);

  if (timeout_ms > 0)
    curl_easy_setopt (easy, CURLOPT_TIMEOUT_MS, (long) timeout_ms);

  /* Method */
  if (strcmp (method, "POST") == 0)
    curl_easy_setopt (easy, CURLOPT_POST, 1L);
  else if (strcmp (method, "PUT") == 0)
    curl_easy_setopt (easy, CURLOPT_UPLOAD, 1L);

  /* Request headers */
  if (!NILP (req_headers_alist))
    {
      Lisp_Object elt;
      EXTERNAL_LIST_LOOP_2 (elt, req_headers_alist)
        {
          char hdr[512];
          Lisp_Object name = XCAR (elt);
          Lisp_Object val  = XCDR (elt);
          snprintf (hdr, sizeof (hdr), "%s: %s",
                    XSTRING_DATA (name), XSTRING_DATA (val));
          req->req_headers = curl_slist_append (req->req_headers, hdr);
        }
      curl_easy_setopt (easy, CURLOPT_HTTPHEADER, req->req_headers);
    }

  /* Body */
  if (!NILP (body_string))
    {
      curl_easy_setopt (easy, CURLOPT_POSTFIELDSIZE,
                        (long) XSTRING_LENGTH (body_string));
      curl_easy_setopt (easy, CURLOPT_COPYPOSTFIELDS,
                        XSTRING_DATA (body_string));
    }

  curl_multi_add_handle (curl_multi, easy);

  return make_opaque_ptr (req);
}

(defmacro await (expr)
  "Evaluate EXPR which must return an opaque coroutine/request handle,
then suspend the current coroutine until the operation completes.
Returns the result value.  Must be called from within a coroutine."
  `(progn
     ,expr
     (async-coroutine-yield)
     (let ((c (async-current-coroutine)))
       (if (and c (not (null (async--coro-error c))))
           (signal 'async-error (list (async--coro-error c)))
         (async--coro-result c)))))

;;; async-let: parallel spawn + join

(defmacro async-let (bindings &rest body)
  "Spawn each binding's expression as a parallel actor, await all, bind results.
Like Promise.all() or asyncio.gather().

\(async-let ((a (http-get url1))
             (b (async-shell-command cmd)))
  (list a b))"
  (declare (indent 1))
  (let ((handles (mapcar (lambda (b) (gensym "handle-")) bindings))
        (vars    (mapcar #'car bindings))
        (exprs   (mapcar #'cadr bindings)))
    `(let ,(mapcar* #'list handles
                    (mapcar (lambda (e)
                              `(actor-spawn nil (lambda (_) ,e) nil))
                            exprs))
       (let ,(mapcar* #'list vars
                      (mapcar (lambda (h) `(await (actor-join ,h)))
                              handles))
         ,@body))))

;;; async-yield: voluntary yield point for CPU-bound coroutines

(defun async-yield ()
  "Voluntarily yield the current coroutine to the scheduler.
Use in long-running CPU-bound loops to avoid blocking other coroutines."
  (async-coroutine-yield))

;;; Deferred: create a pending result that can be resolved later

(defun make-deferred ()
  "Return a deferred object (RESOLVE . REJECT).
Call (funcall (car d) value) to resolve, (funcall (cdr d) error) to reject."
  (let* ((coro (async-current-coroutine))
         (resolve (lambda (val) (actor-send-internal coro val)))
         (reject  (lambda (err) (actor-send-internal coro
                                  (list 'error err)))))
    (cons resolve reject)))

;;; Error types
(define-error 'async-timeout "Async operation timed out")
(define-error 'async-http-error "Async HTTP error")
(define-error 'async-error "Async scheduler error")

(defun http-post (url &rest keys)
  "Async POST to URL.  Returns result to `await': (STATUS HEADERS BODY).
:headers  ALIST   — request headers
:json     OBJECT  — Lisp object to JSON-encode as body (sets Content-Type)
:body     STRING  — raw body string
:on-chunk FUNC    — if non-nil, stream chunks; FUNC called with each chunk string
:timeout  MS      — timeout in milliseconds"

         (body     (plist-get keys :body))
         (on-chunk (plist-get keys :on-chunk))
         (timeout  (plist-get keys :timeout))
         (body-str (cond
                    (json-obj
                     (setq headers
                           (cons '("Content-Type" . "application/json") headers))
                     (json-encode json-obj))
                    (body body)
                    (t nil))))
    (async-http-request-internal url "POST" headers body-str on-chunk timeout)))

    (substring line 6)))

(defun sse-extract-text (chunk)
  "Parse SSE CHUNK string, extract text delta from OpenAI/Anthropic stream format.
Returns concatenated text from all data lines in CHUNK, or nil."
  (let ((result '()))
    (dolist (line (split-string chunk "\n"))

                     (text  (cdr (assq 'content delta))))

            (error nil)))))

      (apply #'concat (nreverse result)))))

  (unless swe-agent-api-key
    (error "swe-agent-api-key is not set"))
  (actor-spawn
   'swe-agent
   (lambda (_)
     (let ((messages (vector `((role . "user") (content . ,task))))
           (done nil))
       (while (not done)
         (let ((chunks '()))
           ;; Stream LLM response
           (with-current-buffer output-buffer
             (goto-char (point-max)))
           (await (http-post swe-agent-api-url
                             :headers (swe-agent-headers)
                             :json `((model . ,swe-agent-model)
                                     (stream . t)
                                     (messages . ,messages))
                             :on-chunk
                             (lambda (chunk)
                               (let ((text (sse-extract-openai-text chunk)))
                                 (when text
                                   (push text chunks)
                                   (with-current-buffer output-buffer
                                     (goto-char (point-max))
                                     (insert text)))))))
           (let* ((response  (apply #'concat (nreverse chunks)))
                  (tool-calls (swe-parse-tool-calls response)))
             (if (null tool-calls)
                 (setq done t)
               (let ((results (swe-execute-tools-parallel tool-calls)))
                 (setq messages
                       (vconcat messages
                                (vector
                                 `((role . "assistant") (content . ,response))
                                 `((role . "tool")
                                   (tool_results
                                    . ,(mapcar (lambda (r)
                                                 `((tool_call_id . ,(car r))
                                                   (content . ,(cdr r))))
                                               results))))))))))))
   nil))

(defun swe-agent (task)
  "Start a SWE-agent session for TASK interactively."
  (interactive "sTask: ")
  (let ((buf (get-buffer-create (format "*swe-agent: %s*"
                                        (substring task 0 (min 40 (length task)))))))
    (switch-to-buffer buf)
    (erase-buffer)
    (swe-agent-loop task buf)
    (message "SWE-agent started in %s" (buffer-name buf))))

(provide 'swe-agent)
```
